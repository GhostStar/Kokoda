cmake_minimum_required(VERSION 3.22)
project(KokodaDesktop VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Test)
get_filename_component(KOKODA_ROOT ../.. ABSOLUTE)
if(WIN32)
  set(KOKODA_FFMPEG_ROOT "" CACHE PATH "Pinned FFmpeg shared SDK")
  find_path(AV_INCLUDE libavcodec/avcodec.h HINTS "${KOKODA_FFMPEG_ROOT}/include" REQUIRED)
  foreach(lib avcodec avutil swscale)
    find_library(${lib}_LIB NAMES ${lib} HINTS "${KOKODA_FFMPEG_ROOT}/lib" REQUIRED)
  endforeach()
  set(AV_LIBRARIES ${avcodec_LIB} ${avutil_LIB} ${swscale_LIB})
  set(PLATFORM_SOURCE ${KOKODA_ROOT}/platforms/windows/client/integration.cpp)
  set(RUNNER ${KOKODA_ROOT}/apps/windows/main.cpp)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale)
  set(AV_LIBRARIES PkgConfig::AV)
  set(PLATFORM_SOURCE ${KOKODA_ROOT}/platforms/linux/client/integration.cpp)
  set(RUNNER ${KOKODA_ROOT}/apps/linux/main.cpp)
endif()
add_library(kokoda-desktop STATIC
  src/core.cpp src/connection.cpp src/session.cpp src/mainwindow.cpp src/viewer.cpp
  ${KOKODA_ROOT}/platforms/common/client/decoder.cpp ${PLATFORM_SOURCE})
target_include_directories(kokoda-desktop PUBLIC src ${KOKODA_ROOT}/platforms/common/client PRIVATE ${AV_INCLUDE})
target_link_libraries(kokoda-desktop PUBLIC Qt6::Widgets PRIVATE ${AV_LIBRARIES})
if(MSVC)
  target_compile_options(kokoda-desktop PRIVATE /utf-8 /W4)
else()
  target_compile_options(kokoda-desktop PRIVATE -Wall -Wextra)
endif()
set_source_files_properties(${KOKODA_ROOT}/packaging/icons/kokoda.png PROPERTIES QT_RESOURCE_ALIAS kokoda.png)
qt_add_resources(kokoda-desktop resources PREFIX / FILES
  locales/en.json locales/zh-CN.json ${KOKODA_ROOT}/packaging/icons/kokoda.png)
if(WIN32)
  enable_language(RC)
  configure_file(${KOKODA_ROOT}/packaging/desktop/kokoda.rc.in kokoda.rc @ONLY)
  set(ICON_RESOURCE ${CMAKE_CURRENT_BINARY_DIR}/kokoda.rc)
endif()
add_executable(Kokoda WIN32 ${RUNNER} ${ICON_RESOURCE})
target_link_libraries(Kokoda PRIVATE kokoda-desktop)
enable_testing()
add_executable(kokoda-desktop-tests ${KOKODA_ROOT}/tests/platform/desktop/client_tests.cpp)
target_compile_definitions(kokoda-desktop-tests PRIVATE KOKODA_FIXTURE_DIR="${KOKODA_ROOT}/tests/fixtures")
target_link_libraries(kokoda-desktop-tests PRIVATE kokoda-desktop Qt6::Test)
add_test(NAME desktop-contract COMMAND kokoda-desktop-tests)
set_tests_properties(desktop-contract PROPERTIES ENVIRONMENT QT_QPA_PLATFORM=offscreen)
install(TARGETS Kokoda RUNTIME DESTINATION bin)
